# Pipeline run: deadline_jitter
# Policy: deadline  Profile: high jitter 20+-20 ms
# Rate: 50 Hz  Duration: 4 s  Out: results/live/deadline_jitter

# 1. Bring-up (done once):
#   python -m uavpipe.doctor        # confirm Nucleo COM port + Pixhawk heartbeat

# 2. Apply link shaping (on the Pi):
   sudo scripts/netem_on.sh wlan0 "delay 20ms 20ms loss 0%"

# 3. Run the gateway (live_log.csv -> results/live/<folder>/live_log.csv):
   python -m uavpipe.live --sensor udp --bind 0.0.0.0:5005 \
       --pixhawk /dev/ttyACM0 --estimator nucleo:/dev/ttyACM1 \
       --hz 50 --policy deadline --seconds 4 --out results/live/deadline_jitter

# 4. Remove link shaping:
   sudo scripts/netem_off.sh wlan0

# 5. Recompute table from raw log:
   python -m uavpipe.analyze_live --folder deadline_jitter
